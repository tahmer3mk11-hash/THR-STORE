<script>
// المنتجات
const products = [
  { id: 1, name: "ببجي 660 UC", price: 38, icon: "🎮" },
  { id: 2, name: "ببجي 1800 UC", price: 95, icon: "🎮" },
  { id: 3, name: "جوجل بلاي 50$", price: 185, icon: "🎁" },
  { id: 4, name: "آيتونز 25$", price: 98, icon: "🍎" },
  { id: 5, name: "نتفليكس شهر", price: 45, icon: "🎬" },
  { id: 6, name: "سبوتيفاي شهر", price: 25, icon: "🎵" },
  { id: 7, name: "ستيم 20$", price: 78, icon: "💻" },
  { id: 8, name: "بلايستيشن Plus", price: 120, icon: "🕹️" }
];

let cart = [];

// عرض المنتجات
const productsDiv = document.getElementById('products');
products.forEach(p => {
  productsDiv.innerHTML += `
    <div class="card">
      <div class="icon">${p.icon}</div>
      <h3>${p.name}</h3>
      <div class="price">${p.price} ر.س</div>
      <button onclick="addToCart(${p.id})">أضف للسلة</button>
    </div>
  `;
});

// إضافة للسلة
function addToCart(id) {
  const product = products.find(p => p.id === id);
  const existing = cart.find(item => item.id === id);
  if (existing) {
    existing.qty++;
  } else {
    cart.push({ ...product, qty: 1 });
  }
  renderCart();
  toggleCart();
}

// عرض السلة
function renderCart() {
  const itemsDiv = document.getElementById('cartItems');
  const countSpan = document.getElementById('count');
  const totalSpan = document.getElementById('total');

  if (cart.length === 0) {
    itemsDiv.innerHTML = '<div class="empty">السلة فارغة</div>';
    countSpan.textContent = 0;
    totalSpan.textContent = '0 ر.س';
    return;
  }

  let html = '';
  let total = 0;
  cart.forEach(item => {
    total += item.price * item.qty;
    html += `
      <div class="cart-item">
        <div>
          <strong>${item.name}</strong><br>
          <small>${item.price} ر.س × ${item.qty}</small>
        </div>
        <span class="remove" onclick="removeFromCart(${item.id})">حذف</span>
      </div>
    `;
  });
  itemsDiv.innerHTML = html;
  countSpan.textContent = cart.reduce((s, i) => s + i.qty, 0);
  totalSpan.textContent = total + ' ر.س';
}

// حذف من السلة
function removeFromCart(id) {
  cart = cart.filter(item => item.id !== id);
  renderCart();
}

// فتح/إغلاق السلة
function toggleCart() {
  document.getElementById('cartPanel').classList.toggle('open');
}

// إتمام الطلب
function checkout() {
  if (cart.length === 0) {
    alert('السلة فارغة!');
    return;
  }
  alert('سيتم تحويلك لبوابة الدفع...\n(هنا تضع رابط بوابة الدفع مثل Moyasar أو Tap)');
  // مثال: window.location.href = 'https://checkout.example.com';
}
